REGRAS DO FIRESTORE — GAB'S (gabs-teste)
=========================================
Onde colar: Firebase Console > Firestore Database > Regras > apague o que estiver lá, cole tudo abaixo > Publicar.

rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    // cardápio: todo mundo lê, só o painel (logado) altera
    match /produtos/{id}     { allow read: if true; allow write: if request.auth != null; }
    match /opcoesLanche/{id} { allow read: if true; allow write: if request.auth != null; }
    match /config/{id}       { allow read: if true; allow write: if request.auth != null; }

    // pedidos: o cliente só cria; o painel (logado) lê, muda status e apaga
    match /orders/{id} {
      allow create: if true;
      allow read, update, delete: if request.auth != null;
    }

    // qualquer outra coisa (caixa, histórico...): só logado
    match /{document=**} { allow read, write: if request.auth != null; }
  }
}
